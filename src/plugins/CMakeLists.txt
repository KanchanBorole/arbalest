FILE(GLOB DIRECTORY_LIST RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/*")

FOREACH(LIST_ENTRY ${DIRECTORY_LIST})
    IF(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${LIST_ENTRY}/CMakeLists.txt")
        MESSAGE(STATUS "Adding plugin from ${CMAKE_CURRENT_SOURCE_DIR}/${LIST_ENTRY}")
        ADD_SUBDIRECTORY("${CMAKE_CURRENT_SOURCE_DIR}/${LIST_ENTRY}")
    ENDIF(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${LIST_ENTRY}/CMakeLists.txt")
ENDFOREACH(LIST_ENTRY ${DIRECTORY_LIST})
